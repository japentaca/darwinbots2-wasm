 *.eye5 0 >
*.in7 *.out7 !=

 *.refvelup .up store

*.eye5 0 =
*.in7 *.out7 =
or
start
157 314 rnd sub .aimdx store
10 .up store
stop

cond
 *.eye5 0 >
*.in7 *.out7 !=
start
*.refpoison 0 = *.refshell 0 > and
-1 .shoot store
not
-6 .shoot store
true

stop

*.waste 50 >
